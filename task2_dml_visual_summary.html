3 ways to insert data

-- 1. Full row (all columns) INSERT INTO customers (first_name, last_name, email, phone) VALUES ('Arjun', 'Sharma', 'a@b.com', '9876543210'); -- 2. Partial columns — phone gets NULL, created_at gets DEFAULT INSERT INTO customers (first_name, last_name, email) VALUES ('Divya', 'Menon', 'd@m.com'); -- 3. INSERT … SELECT (populate summary from live data) INSERT INTO order_summary (customer_id, full_name, order_count, total_spent) SELECT c.customer_id, CONCAT(c.first_name, ' ', c.last_name), COUNT(o.order_id), COALESCE(SUM(o.total_amount), 0) FROM customers c LEFT JOIN orders o ON c.customer_id = o.customer_id GROUP BY c.customer_id;

NULL handling techniques

-- IS NULL — find missing phone numbers SELECT customer_id, first_name, COALESCE(phone, 'Not provided') AS phone FROM customers WHERE phone IS NULL; -- IFNULL — short form of COALESCE for one fallback IFNULL(CAST(paid_at AS CHAR), 'Not yet paid') -- CASE WHEN for descriptive NULL labels CASE WHEN description IS NULL THEN '⚠ Missing' ELSE description END -- NULL audit — count nulls per column SELECT 'customers.phone', COUNT(*) - COUNT(phone) FROM customers UNION ALL SELECT 'payments.paid_at', COUNT(*) - COUNT(paid_at) FROM payments;

UPDATE patterns

-- Single row UPDATE customers SET phone = '9111222333' WHERE customer_id = 6; -- Multiple rows matching a condition UPDATE products SET description = 'No description yet.' WHERE description IS NULL; -- Recalculate total_amount from child rows (correlated subquery) UPDATE orders o SET total_amount = ( SELECT COALESCE(SUM(oi.quantity * oi.unit_price), 0) FROM order_items oi WHERE oi.order_id = o.order_id ); -- UPDATE with JOIN — deduct stock for an order UPDATE products p JOIN order_items oi ON oi.product_id = p.product_id SET p.stock_qty = p.stock_qty - oi.quantity WHERE oi.order_id = 6;

DELETE & ROLLBACK

-- Safe delete — always use WHERE DELETE FROM reviews WHERE review_id = 5; -- Batch delete — remove cancelled orders DELETE FROM order_items WHERE order_id IN ( SELECT order_id FROM orders WHERE status = 'cancelled' ); DELETE FROM orders WHERE status = 'cancelled'; -- ON DELETE CASCADE removes payment row automatically ↑ -- ROLLBACK — undo an accidental deletion START TRANSACTION; DELETE FROM products WHERE stock_qty < 30; ROLLBACK; -- all rows restored, nothing committed
